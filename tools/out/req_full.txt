_DOSE_UNITS   = {&quot;mg&quot;, &quot;毫克&quot;}
WEIGHT_MIN_KG, WEIGHT_MAX_KG = 20.0, 300.0
NE_LABEL_BASIS = &quot;base&quot;
_LABEL_TO_BASE = {&quot;base&quot;: 1.0, &quot;salt&quot;: 0.5}
def ne_ugkgmin(action, weight_kg):
    dq = []
    if NE_LABEL_BASIS not in _LABEL_TO_BASE:
        return None, [&quot;NE_LABEL_BASIS 未配置，拒绝计算&quot;]
    if (action.get(&quot;doseUnit&quot;) or &quot;&quot;).strip() not in _DOSE_UNITS:
        return None, [&quot;dose 单位不在白名单&quot;]
    if (action.get(&quot;liquidUnit&quot;) or &quot;&quot;).strip() not in _LIQUID_UNITS:
        return None, [&quot;liquidAmount 单位不在白名单&quot;]
    liquid_ml = float(action.get(&quot;liquidAmount&quot;) or 0)
    if liquid_ml &lt;= 0:
        return None, [&quot;liquidAmount 为 0 或缺失，无法算浓度&quot;]
    if not (WEIGHT_MIN_KG &lt;= (weight_kg or 0) &lt;= WEIGHT_MAX_KG):
        return None, [&quot;体重超出 20 至 300 kg&quot;]
    conc_ug_ml = float(action[&quot;dose&quot;]) / liquid_ml * 1000.0
    val = conc_ug_ml * float(action[&quot;speed&quot;]) / 60.0 / weight_kg * _LABEL_TO_BASE[NE_LABEL_BASIS]
    if val &gt; 2.0:
        dq.append(&quot;NE 剂量异常高，需人工复核&quot;)
    return val, dq
